  <!-- Logica și conexiunea cu Firebase -->
  <script type="module">
    import { initializeApp } from "https://www.gstatic.com/firebasejs/10.5.0/firebase-app.js";
    import { getFirestore, collection, onSnapshot, doc, updateDoc, addDoc, serverTimestamp, query, orderBy } from "https://www.gstatic.com/firebasejs/10.5.0/firebase-firestore.js";

    // ==========================================
    // 2. MODIFICĂ AICI: DATELE TALE FIREBASE
    // Înlocuiește textul din interiorul ghilimelelor cu datele din consola ta Firebase
    // ==========================================
    const firebaseConfig = {
      apiKey: "AICI_PUI_API_KEY",
      authDomain: "nume-proiect.firebaseapp.com",
      projectId: "nume-proiect",
      storageBucket: "nume-proiect.appspot.com",
      messagingSenderId: "123456789",
      appId: "1:123456789:web:abcdef"
    };

    // Inițializare aplicație
    const app = initializeApp(firebaseConfig);
    const db = getFirestore(app);
    
    let currentItemId = '';
    let currentItemStock = 0;
    let currentItemNameText = '';

    // Afișează stocurile în timp real
    onSnapshot(collection(db, "stocuri"), (snapshot) => {
      const list = document.getElementById("inventoryList");
      list.innerHTML = '';
      snapshot.forEach((doc) => {
        const item = doc.data();
        list.innerHTML += `
          <div class="card item-card">
            <div>
              <h3 class="item-name">${item.nume}</h3>
              <p style="margin:0; color:#555;">Stoc: <strong>${item.cantitate}</strong> buc.</p>
            </div>
            <div>
              <button class="btn-history" onclick="openTransactionModal('${doc.id}', '${item.nume}', ${item.cantitate})">Modifică</button>
            </div>
          </div>
        `;
      });
    });

    // Afișează istoricul în timp real
    const q = query(collection(db, "istoric"), orderBy("data", "desc"));
    onSnapshot(q, (snapshot) => {
      const tbody = document.getElementById("historyTableBody");
      tbody.innerHTML = '';
      snapshot.forEach((doc) => {
        const tx = doc.data();
        const dateStr = tx.data ? tx.data.toDate().toLocaleString('ro-RO', {day:'2-digit', month:'2-digit', hour:'2-digit', minute:'2-digit'}) : 'Acum';
        const color = tx.tip === 'add' ? '#28a745' : '#dc3545';
        const sign = tx.tip === 'add' ? '+' : '-';
        
        tbody.innerHTML += `
          <tr>
            <td>${dateStr}</td>
            <td>${tx.electrician}</td>
            <td>${tx.material}</td>
            <td style="color: ${color}; font-weight: bold;">${sign}${tx.cantitate}</td>
          </tr>
        `;
      });
    });

    // Creare material nou
    window.addNewItem = async function() {
      const nume = document.getElementById("newItemName").value;
      const cantitate = parseInt(document.getElementById("newItemQty").value);
      if (nume && cantitate >= 0) {
        await addDoc(collection(db, "stocuri"), { nume: nume, cantitate: cantitate });
        document.getElementById("newItemName").value = '';
        document.getElementById("newItemQty").value = '';
      } else {
        alert("Te rog introdu un nume și o cantitate validă.");
      }
    };

    // Deschidere fereastră tranzacție
    window.openTransactionModal = function(id, name, stock) {
      currentItemId = id;
      currentItemNameText = name;
      currentItemStock = stock;
      document.getElementById("modalItemName").innerText = name;
      document.getElementById("transactionQty").value = '';
      document.getElementById("electricianName").value = '';
      document.getElementById("transactionModal").style.display = "flex";
    };

    // Confirmare Adăugare sau Extragere
    window.submitTransaction = async function(tip) {
      const qty = parseInt(document.getElementById("transactionQty").value);
      const electrician = document.getElementById("electricianName").value;

      if (!electrician) { alert("Selectează electricianul!"); return; }
      if (!qty || qty <= 0) { alert("Introdu o cantitate mai mare ca zero!"); return; }

      let newStock = currentItemStock;
      if (tip === 'add') {
        newStock += qty;
      } else if (tip === 'remove') {
        if (qty > currentItemStock) { alert("Atenție! Nu poți extrage mai mult decât ai pe stoc."); return; }
        newStock -= qty;
      }

      // Actualizăm stocul în Firebase
      await updateDoc(doc(db, "stocuri", currentItemId), { cantitate: newStock });

      // Salvăm mișcarea în istoric
      await addDoc(collection(db, "istoric"), {
        material: currentItemNameText,
        electrician: electrician,
        cantitate: qty,
        tip: tip,
        data: serverTimestamp()
      });

      closeModal('transactionModal');
    };

    // Funcția pentru motorul de căutare
    window.filterItems = function() {
      const input = document.getElementById("searchInput").value.toLowerCase();
      const cards = document.getElementsByClassName("item-card");
      for (let i = 0; i < cards.length; i++) {
        const name = cards[i].getElementsByClassName("item-name")[0].innerText.toLowerCase();
        cards[i].style.display = name.includes(input) ? "flex" : "none";
      }
    };

    // Închidere/Deschidere ferestre (Modals)
    window.openHistory = function() { document.getElementById("historyModal").style.display = "flex"; };
    window.closeModal = function(id) { document.getElementById(id).style.display = "none"; };
  </script>
